function getAvailableTextHeight(){
const style = getComputedStyle(layoutBox);
const gap = parseFloat(style.gap) || 0;
let used = 0;

Array.from(layoutBox.children).forEach(child=>{
if(child === previewText) return;
if(getComputedStyle(child).display === "none") return;
used += child.offsetHeight + gap;
});

return layoutBox.clientHeight - used - 12;
}

function safeFit(){
applyControlStyles();

let size = parseInt(fontSize.value);
let minSize = 16;
let maxHeight = getAvailableTextHeight();

previewText.style.fontSize = size + "px";

while(previewText.scrollHeight > maxHeight && size > minSize){
size--;
previewText.style.fontSize = size + "px";
}

/* لا نكبّر النص تلقائيًا أبدًا */
}

previewText.style.wordSpacing = wordSpacing.value + "px";
previewText.style.padding = textPadding.value + "px";

layoutBox.style.gap = gapRange.value + "px";
}
